function runDailyBackup() {
  try {

    // ১. ফায়ারবেসে লগ-ইন করে Token নেওয়া

    for (let i = 0; i < COLLECTIONS.length; i++) {
      const colName = COLLECTIONS[i];
      Logger.log("Fetching: " + colName);
      fullData[colName] = fetchCollection(colName, idToken);
    }

    // ৩. JSON ফরম্যাটে ডেটা প্রস্তুত করা

    const dateStr = Utilities.formatDate(new Date(), "Asia/Dhaka", "yyyy-MM-dd");
    const fileName = "Maa_Motors_ERP_Backup_" + dateStr + ".json";

    // ৪. গুগল ড্রাইভে 'ERP_Auto_Backups' ফোল্ডার তৈরি বা খোঁজা
    const folderName = "ERP_Auto_Backups";
    let folder;
    const folders = DriveApp.getFoldersByName(folderName);

  } catch (e) {
    Logger.log("Error during backup: " + e.toString());
  }
}

// ফায়ারবেস অথেন্টিকেশন (Login)
function getAuthToken() {
  const url = "https://identitytoolkit.googleapis.com/v1/accounts:signInWithPassword?key=" + API_KEY;
  const payload = {
    email: ADMIN_EMAIL,
    password: ADMIN_PASSWORD,
    returnSecureToken: true
  };
  
  const options = {
    method: "post",
    contentType: "application/json",
    payload: JSON.stringify(payload),
    muteHttpExceptions: true
  };
  
  const response = UrlFetchApp.fetch(url, options);
  const data = JSON.parse(response.getContentText());
  
  return data.idToken || null;
}

// ফায়ারবেস ক্লাউড ফায়ারস্টোর থেকে ডাটা পড়া (REST API)
function fetchCollection(collectionName, token) {
  const baseUrl = "https://firestore.googleapis.com/v1/projects/" + PROJECT_ID + "/databases/(default)/documents/" + collectionName + "?pageSize=1000";
  const options = {
    method: "get",
    headers: {
      "Authorization": "Bearer " + token
    },
    muteHttpExceptions: true
  };
  
  let allDocuments = [];
  let nextPageToken = null;
  
  do {
    let url = baseUrl;
    if (nextPageToken) {
      url += "&pageToken=" + encodeURIComponent(nextPageToken);
    }
    
    const response = UrlFetchApp.fetch(url, options);
    const data = JSON.parse(response.getContentText());
    
    if (data.documents && data.documents.length > 0) {
      // Clean up Firestore REST format to simple JSON
      data.documents.forEach(doc => {
        let id = doc.name.split('/').pop();
        let cleanedData = parseFirestoreFormat(doc.fields || {});
        allDocuments.push({ id: id, ...cleanedData });
      });
    }
    
    nextPageToken = data.nextPageToken;
    
  } while (nextPageToken);
  
  return allDocuments;
}

// Firestore REST API ফরম্যাট থেকে সাধারণ JSON এ রূপান্তর
function parseFirestoreFormat(fields) {
  const result = {};
  for (let key in fields) {
    let type = Object.keys(fields[key])[0];
    let value = fields[key][type];
    
    if (type === 'integerValue') result[key] = parseInt(value, 10);
    else if (type === 'doubleValue') result[key] = parseFloat(value);
    else if (type === 'booleanValue') result[key] = value;
    else if (type === 'stringValue') result[key] = value;
    else if (type === 'timestampValue') result[key] = value;
    else if (type === 'arrayValue') {
      result[key] = (value.values || []).map(v => parseFirestoreFormat({ temp: v }).temp);
    }
    else if (type === 'mapValue') {
      result[key] = parseFirestoreFormat(value.fields || {});
    }
    else if (type === 'nullValue') result[key] = null;
    else result[key] = value; // Default fallback
  }
  return result;
}
